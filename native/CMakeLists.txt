cmake_minimum_required(VERSION 3.16)
project(quorum_signer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT CBMPC_SOURCE)
  message(FATAL_ERROR "Set CBMPC_SOURCE to pinned upstream checkout")
endif()
add_subdirectory(${CBMPC_SOURCE} cbmpc)
add_executable(quorum-signer signer.cpp)
target_link_libraries(quorum-signer PRIVATE cbmpc)
target_include_directories(quorum-signer PRIVATE ${CBMPC_SOURCE}/include)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  target_compile_options(quorum-signer PRIVATE -march=armv8-a+crypto)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(quorum-signer PRIVATE pthread dl rt)
endif()
